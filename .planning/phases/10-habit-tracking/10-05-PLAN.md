import Foundation
import SwiftData

/// Local record of a habit completion for offline-first UX.
/// Synced with backend via POST /habits/{id}/complete.
@Model
class HabitCompletion {
    @Attribute(.unique) var id: UUID
    var habitId: UUID
    var completedAt: Date
    var userTimezone: String
    var syncedToBackend: Bool

    init(
        id: UUID = UUID(),
        habitId: UUID,
        completedAt: Date = Date(),
        userTimezone: String = TimeZone.current.identifier,
        syncedToBackend: Bool = false
    ) {
        self.id = id
        self.habitId = habitId
        self.completedAt = completedAt
        self.userTimezone = userTimezone
        self.syncedToBackend = syncedToBackend
    }
}
```

// Update HabitDTO
struct HabitDTO: Codable, Identifiable {
    let id: UUID
    let videoId: UUID?
    let title: String
    let frequency: String
    let currentStreak: Int
    let longestStreak: Int
    let reminderTime: String?  // "HH:MM:SS" format from backend
    let reminderDays: [Int]?
    let isActive: Bool
    let createdAt: Date

    enum CodingKeys: String, CodingKey {
        case id
        case videoId = "video_id"
        case title
        case frequency
        case currentStreak = "current_streak"
        case longestStreak = "longest_streak"
        case reminderTime = "reminder_time"
        case reminderDays = "reminder_days"
        case isActive = "is_active"
        case createdAt = "created_at"
    }

    var frequencyEnum: HabitFrequency {
        HabitFrequency(rawValue: frequency) ?? .daily
    }

    /// Parse reminderTime string into DateComponents (hour, minute)
    var reminderTimeComponents: DateComponents? {
        guard let timeString = reminderTime else { return nil }
        let parts = timeString.split(separator: ":")
        guard parts.count >= 2,
              let hour = Int(parts[0]),
              let minute = Int(parts[1]) else { return nil }
        return DateComponents(hour: hour, minute: minute)
    }
}

// Add HabitUpdateRequest
struct HabitUpdateRequest: Codable {
    var title: String?
    var frequency: String?
    var reminderTime: String?  // "HH:MM" format
    var reminderDays: [Int]?
    var isActive: Bool?

    enum CodingKeys: String, CodingKey {
        case title
        case frequency
        case reminderTime = "reminder_time"
        case reminderDays = "reminder_days"
        case isActive = "is_active"
    }
}

// Add HabitCompletionRequest and Response
struct HabitCompletionRequest: Codable {
    let userTimezone: String

    enum CodingKeys: String, CodingKey {
        case userTimezone = "user_timezone"
    }

    init(timezone: String = TimeZone.current.identifier) {
        self.userTimezone = timezone
    }
}

struct HabitCompletionResponse: Codable {
    let id: UUID
    let habitId: UUID
    let completedAt: Date

    enum CodingKeys: String, CodingKey {
        case id
        case habitId = "habit_id"
        case completedAt = "completed_at"
        case currentStreak = "current_streak"
        case longestStreak = "longest_streak"
    }
}

// MARK: - Habits

/// Update a habit (partial update).
func updateHabit(habitId: UUID, request: HabitUpdateRequest) async throws -> HabitDTO {
    let url = baseURL.appendingPathComponent("habits/\(habitId.uuidString)")
    var urlRequest = URLRequest(url: url)
    urlRequest.httpMethod = "PATCH"
    urlRequest.setValue("application/json", forHTTPHeaderField: "Content-Type")

    try await addAuthHeader(to: &urlRequest)
    urlRequest.httpBody = try encoder.encode(request)

    let (data, response) = try await URLSession.shared.data(for: urlRequest)
    try handleHTTPResponse(response)

    return try decoder.decode(HabitDTO.self, from: data)
}

/// Record a habit completion.
func completeHabit(habitId: UUID, timezone: String = TimeZone.current.identifier) async throws -> HabitCompletionResponse {
    let url = baseURL.appendingPathComponent("habits/\(habitId.uuidString)/complete")
    var urlRequest = URLRequest(url: url)
    urlRequest.httpMethod = "POST"
    urlRequest.setValue("application/json", forHTTPHeaderField: "Content-Type")

    try await addAuthHeader(to: &urlRequest)

    let request = HabitCompletionRequest(timezone: timezone)
    urlRequest.httpBody = try encoder.encode(request)

    let (data, response) = try await URLSession.shared.data(for: urlRequest)
    try handleHTTPResponse(response)

    return try decoder.decode(HabitCompletionResponse.self, from: data)
}

/// Get habit completions for a habit (for calendar display).
func getHabitCompletions(habitId: UUID) async throws -> [HabitCompletionResponse] {
    let url = baseURL.appendingPathComponent("habits/\(habitId.uuidString)/completions")
    var urlRequest = URLRequest(url: url)

    try await addAuthHeader(to: &urlRequest)

    let (data, response) = try await URLSession.shared.data(for: urlRequest)
    try handleHTTPResponse(response)

    return try decoder.decode([HabitCompletionResponse].self, from: data)
}
```

Note: The getHabitCompletions endpoint will be added to backend in a later plan if needed. For now, completions are tracked locally and synced to backend.
