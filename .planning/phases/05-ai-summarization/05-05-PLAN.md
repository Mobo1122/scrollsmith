class UpdateSummaryRequest(BaseModel):
    """Request to manually update video summary fields.

    All fields are optional - only provided fields will be updated.
    This allows partial updates (e.g., edit only tags, or only bullets).
    """
    bullets: list[str] | None = Field(
        None,
        min_length=1,
        max_length=15,
        description="Updated bullet points (3-15 bullets)"
    )
    tags: list[str] | None = Field(
        None,
        max_length=10,
        description="Updated tags (up to 10 tags)"
    )
    steps: dict | None = Field(
        None,
        description="Updated step checklist (must match StepChecklist schema structure)"
    )
    cards: dict | None = Field(
        None,
        description="Updated cards (must match CardsSummary schema structure)"
    )

    class Config:
        json_schema_extra = {
            "example": {
                "bullets": [
                    "Key insight 1",
                    "Key insight 2",
                    "Key insight 3"
                ],
                "tags": ["productivity", "habits", "mindset"]
            }
        }
```

This schema:
- Supports partial updates (all fields optional)
- Validates bullets array length (prevent empty or excessive bullets)
- Validates tags array length (prevent tag spam)
- Accepts steps/cards as dicts (will be validated against Pydantic schemas before saving)
- Follows FastAPI best practices for PATCH endpoints (partial updates)

@router.patch("/{video_id}/summary", response_model=VideoResponse)
async def update_summary(
    video_id: UUID,
    request: UpdateSummaryRequest,
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_db),
) -> Video:
    """Manually update video summary fields.

    Allows users to edit AI-generated summaries and tags. Supports partial
    updates - only provided fields will be modified.

    Args:
        video_id: Video to update
        request: Fields to update (all optional)

    Returns:
        Updated video record

    Raises:
        404: Video not found or not owned by user
        400: Invalid summary structure (steps/cards don't match schema)
    """
    # Fetch video
    result = await db.execute(
        select(Video).where(
            Video.id == video_id,
            Video.user_id == current_user.id
        )
    )
    video = result.scalar_one_or_none()

    if video is None:
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND,
            detail="Video not found"
        )

    # Track which fields are being updated
    updates = []

    # Update bullets (stored as JSON with tags)
    if request.bullets is not None:
        if len(request.bullets) < 3:
            raise HTTPException(
                status_code=status.HTTP_400_BAD_REQUEST,
                detail="Bullets must contain at least 3 items"
            )

        # Reconstruct BulletSummary structure
        from app.schemas.summary import BulletSummary
        bullet_summary = BulletSummary(
            bullets=request.bullets,
            tags=request.tags if request.tags is not None else video.tags or []
        )
        video.summary_bullets = bullet_summary.model_dump_json()
        updates.append("bullets")

    # Update tags independently (if bullets not updated)
    elif request.tags is not None:
        video.tags = request.tags
        # Also update bullets JSON if it exists (to keep tags in sync)
        if video.summary_bullets:
            import json
            bullet_data = json.loads(video.summary_bullets)
            bullet_data["tags"] = request.tags
            video.summary_bullets = json.dumps(bullet_data)
        updates.append("tags")

    # Update step checklist (Pro tier)
    if request.steps is not None:
        try:
            # Validate against StepChecklist schema
            step_checklist = StepChecklist(**request.steps)
            video.summary_steps = step_checklist.model_dump_json()
            updates.append("steps")
        except Exception as e:
            raise HTTPException(
                status_code=status.HTTP_400_BAD_REQUEST,
                detail=f"Invalid step checklist structure: {str(e)}"
            )

    # Update cards (Pro tier)
    if request.cards is not None:
        try:
            # Validate against CardsSummary schema
            cards_summary = CardsSummary(**request.cards)
            video.summary_cards = cards_summary.model_dump_json()
            updates.append("cards")
        except Exception as e:
            raise HTTPException(
                status_code=status.HTTP_400_BAD_REQUEST,
                detail=f"Invalid cards structure: {str(e)}"
            )

    if not updates:
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail="No fields to update. Provide at least one field: bullets, tags, steps, or cards."
        )

    # Save changes
    await db.commit()
    await db.refresh(video)

    logger.info(f"Updated summary for video {video_id}: {', '.join(updates)}")

    return video
```

Key implementation details:
- Verifies video ownership (user_id match)
- Supports partial updates (only updates provided fields)
- Validates bullets length (minimum 3, catches empty arrays)
- Validates steps/cards against Pydantic schemas before saving
- Updates tags in both video.tags AND video.summary_bullets JSON (keeps them in sync)
- Returns full Video object (VideoResponse schema already handles all fields)
- Logs which fields were updated for debugging
