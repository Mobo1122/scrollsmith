import SwiftUI

/// Usage indicator pill for app header.
///
/// CONTEXT.md decisions:
/// - Progress ring with number inside
/// - Color changes: green -> yellow -> red
/// - Tappable to open details sheet
struct UsagePillView: View {
    let usage: UsageState?
    @Binding var showDetails: Bool

    var body: some View {
        Button {
            showDetails = true
        } label: {
            HStack(spacing: 6) {
                // Progress ring
                ZStack {
                    Circle()
                        .stroke(Color.gray.opacity(0.2), lineWidth: 3)
                        .frame(width: 28, height: 28)

                    Circle()
                        .trim(from: 0, to: usage?.usageProgress ?? 0)
                        .stroke(ringColor, style: StrokeStyle(lineWidth: 3, lineCap: .round))
                        .frame(width: 28, height: 28)
                        .rotationEffect(.degrees(-90))

                    // Count inside
                    if let usage = usage {
                        Text("\(usage.videosUsed)")
                            .font(.system(size: 10, weight: .bold))
                            .foregroundColor(ringColor)
                    }
                }

                // Label for non-Pro
                if let usage = usage, !usage.isPro {
                    Text("/ \(usage.videosLimit)")
                        .font(.caption2)
                        .foregroundStyle(.secondary)
                }

                // Pro badge
                if usage?.isPro == true {
                    Image(systemName: "crown.fill")
                        .font(.caption)
                        .foregroundColor(.yellow)
                }
            }
            .padding(.horizontal, 8)
            .padding(.vertical, 4)
            .background(
                Capsule()
                    .fill(Color(.systemGray6))
            )
        }
        .buttonStyle(.plain)
    }

    private var ringColor: Color {
        guard let usage = usage else { return .gray }
        switch usage.usageColor {
        case .green: return .green
        case .yellow: return .yellow
        case .red: return .red
        }
    }
}

#Preview("Free - Low Usage") {
    UsagePillView(
        usage: UsageState(
            videosUsed: 3,
            videosLimit: 10,
            resetDate: Date(),
            isPro: false,
            canCreateVideo: true
        ),
        showDetails: .constant(false)
    )
}

#Preview("Free - At Limit") {
    UsagePillView(
        usage: UsageState(
            videosUsed: 10,
            videosLimit: 10,
            resetDate: Date(),
            isPro: false,
            canCreateVideo: false
        ),
        showDetails: .constant(false)
    )
}

#Preview("Pro") {
    UsagePillView(
        usage: UsageState(
            videosUsed: 25,
            videosLimit: -1,
            resetDate: Date(),
            isPro: true,
            canCreateVideo: true
        ),
        showDetails: .constant(false)
    )
}

import SwiftUI

/// Usage details sheet with upgrade option.
///
/// CONTEXT.md decisions:
/// - Current period only (no history charts)
/// - Usage count, reset date, upgrade button
struct UsageDetailSheet: View {
    let usage: UsageState?
    @State private var showPaywall = false
    @Environment(\.dismiss) private var dismiss

    var body: some View {
        NavigationStack {
            VStack(spacing: 24) {
                // Large usage display
                usageDisplay

                // Reset info
                resetInfo

                // Upgrade section (free users only)
                if usage?.isPro != true {
                    upgradeSection
                }

                Spacer()
            }
            .padding()
            .navigationTitle("Usage & Limits")
            .navigationBarTitleDisplayMode(.inline)
            .toolbar {
                ToolbarItem(placement: .topBarTrailing) {
                    Button("Done") { dismiss() }
                }
            }
            .sheet(isPresented: $showPaywall) {
                ScrollsmithPaywallView()
            }
        }
        .presentationDetents([.medium])
    }

    private var usageDisplay: some View {
        VStack(spacing: 8) {
            if let usage = usage {
                ZStack {
                    Circle()
                        .stroke(Color.gray.opacity(0.2), lineWidth: 8)
                        .frame(width: 120, height: 120)

                    Circle()
                        .trim(from: 0, to: usage.usageProgress)
                        .stroke(ringColor, style: StrokeStyle(lineWidth: 8, lineCap: .round))
                        .frame(width: 120, height: 120)
                        .rotationEffect(.degrees(-90))

                    VStack(spacing: 2) {
                        Text("\(usage.videosUsed)")
                            .font(.system(size: 36, weight: .bold))
                        if !usage.isPro {
                            Text("of \(usage.videosLimit)")
                                .font(.caption)
                                .foregroundStyle(.secondary)
                        } else {
                            Text("videos")
                                .font(.caption)
                                .foregroundStyle(.secondary)
                        }
                    }
                }

                Text(usage.isPro ? "Pro - Unlimited" : "Free Plan")
                    .font(.headline)
                    .foregroundStyle(usage.isPro ? .primary : .secondary)
            }
        }
    }

    private var resetInfo: some View {
        Group {
            if let usage = usage, !usage.isPro {
                HStack {
                    Image(systemName: "arrow.clockwise")
                        .foregroundStyle(.secondary)
                    Text("Resets on \(usage.resetDate.formatted(.dateTime.month().day()))")
                        .font(.subheadline)
                        .foregroundStyle(.secondary)
                }
            }
        }
    }

    private var upgradeSection: some View {
        VStack(spacing: 12) {
            Text("Get unlimited videos with Pro")
                .font(.subheadline)

            Button {
                showPaywall = true
            } label: {
                Text("Upgrade to Pro")
                    .font(.headline)
                    .frame(maxWidth: .infinity)
                    .padding()
                    .background(
                        RoundedRectangle(cornerRadius: 12)
                            .fill(.linearGradient(
                                colors: [.purple, .blue],
                                startPoint: .leading,
                                endPoint: .trailing
                            ))
                    )
                    .foregroundColor(.white)
            }
        }
        .padding()
        .background(
            RoundedRectangle(cornerRadius: 12)
                .fill(Color(.systemGray6))
        )
    }

    private var ringColor: Color {
        guard let usage = usage else { return .gray }
        switch usage.usageColor {
        case .green: return .green
        case .yellow: return .yellow
        case .red: return .red
        }
    }
}

#Preview {
    Text("Trigger")
        .sheet(isPresented: .constant(true)) {
            UsageDetailSheet(usage: UsageState(
                videosUsed: 7,
                videosLimit: 10,
                resetDate: Calendar.current.date(from: DateComponents(year: 2026, month: 2, day: 1))!,
                isPro: false,
                canCreateVideo: true
            ))
        }
}
