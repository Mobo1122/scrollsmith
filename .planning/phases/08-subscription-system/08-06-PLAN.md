import SwiftUI
import RevenueCat
import RevenueCatUI

/// Main paywall view using RevenueCatUI.
///
/// CONTEXT.md decisions:
/// - Free vs Pro comparison table
/// - FOMO-driven messaging
/// - Annual pre-selected with "Best value" badge
/// - 3-day trial displayed
struct ScrollsmithPaywallView: View {
    @Environment(\.dismiss) private var dismiss

    var body: some View {
        PaywallView()
            .onPurchaseCompleted { customerInfo in
                // Purchase successful
                Task {
                    await SubscriptionService.shared.refreshEntitlements()
                }
                dismiss()
            }
            .onRestoreCompleted { customerInfo in
                // Restore successful
                Task {
                    await SubscriptionService.shared.refreshEntitlements()
                }
                if customerInfo.entitlements["pro"]?.isActive == true {
                    dismiss()
                }
            }
    }
}

#Preview {

}
```

- Trial eligibility check
- Purchase flow
- Loading states
- Error handling

Dashboard configuration controls the actual layout and copy (remote config).

/// Purchase a package from the current offering.
func purchase(package: Package) async throws -> Bool {
    guard isConfigured else {
        throw SubscriptionError.notConfigured
    }

    let result = try await Purchases.shared.purchase(package: package)
    await refreshEntitlements()
    return result.customerInfo.entitlements["pro"]?.isActive == true
}

/// Restore previous purchases.
func restorePurchases() async throws -> Bool {
    guard isConfigured else {
        throw SubscriptionError.notConfigured
    }

    let customerInfo = try await Purchases.shared.restorePurchases()
    await refreshEntitlements()
    return customerInfo.entitlements["pro"]?.isActive == true
}

/// Get current offerings.
func getOfferings() async throws -> Offerings? {
    guard isConfigured else {
        throw SubscriptionError.notConfigured
    }
    return try await Purchases.shared.offerings()
}

enum SubscriptionError: Error {
    case notConfigured
    case purchaseFailed
    case restoreFailed
}
```

These methods supplement RevenueCatUI for programmatic access if needed.
