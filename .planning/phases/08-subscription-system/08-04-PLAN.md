  <action>
Create ios/Scrollsmith/Services/SubscriptionService.swift:

```swift
import Foundation
import RevenueCat

/// Wrapper around RevenueCat SDK for subscription management.
@MainActor
@Observable
final class SubscriptionService {
    static let shared = SubscriptionService()

    private(set) var isPro = false
    private(set) var isConfigured = false

    private init() {}

    /// Configure RevenueCat with authenticated user's ID.
    /// Call ONLY after authentication succeeds.
    func configure(userId: UUID) {
        guard !isConfigured else { return }

        #if DEBUG
        Purchases.logLevel = .debug
        #endif

        Purchases.configure(
            withAPIKey: "appl_YOUR_PUBLIC_KEY", // TODO: Replace with actual key
            appUserID: userId.uuidString.lowercased()
        )
        isConfigured = true

        Task {
            await refreshEntitlements()
        }
    }

    /// Refresh entitlement status from RevenueCat.
    func refreshEntitlements() async {
        guard isConfigured else { return }

        do {
            let customerInfo = try await Purchases.shared.customerInfo()
            isPro = customerInfo.entitlements["pro"]?.isActive == true
        } catch {
            print("Failed to fetch customer info: \(error)")
            isPro = false
        }
    }

    /// Check if user has Pro access (cached, fast).
    var hasProAccess: Bool {
        isPro
    }
}
```

Key decisions per RESEARCH.md:
- Configure ONLY after auth (never with anonymous ID)
- Use lowercase UUID string for app_user_id (matches backend)
- Check "pro" entitlement isActive (handles expiry, grace period automatically)
