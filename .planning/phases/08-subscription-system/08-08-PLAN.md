```swift
import SwiftUI

struct SettingsView: View {
    @State private var isRestoring = false
    @State private var restoreResult: RestoreResult?
    @State private var showPaywall = false

    var body: some View {
        Form {
            // Subscription section
            Section("Subscription") {
                // Current status
                HStack {
                    Text("Status")
                    Spacer()
                    Text(SubscriptionService.shared.isPro ? "Pro" : "Free")
                        .foregroundStyle(.secondary)
                }

                // Upgrade button (free users)
                if !SubscriptionService.shared.isPro {
                    Button("Upgrade to Pro") {
                        showPaywall = true
                    }
                }

                // Restore purchases
                Button {
                    Task {
                        await restorePurchases()
                    }
                } label: {
                    HStack {
                        Text("Restore Purchases")
                        Spacer()
                        if isRestoring {
                            ProgressView()
                        }
                    }
                }
                .disabled(isRestoring)
            }

            // Account section
            Section("Account") {
                // ... existing account settings
            }
        }
        .navigationTitle("Settings")
        .sheet(isPresented: $showPaywall) {
            ScrollsmithPaywallView()
        }
        .alert("Restore Purchases", isPresented: .constant(restoreResult != nil)) {
            Button("OK") { restoreResult = nil }
        } message: {
            Text(restoreResult?.message ?? "")
        }
    }

    private func restorePurchases() async {
        isRestoring = true
        do {
            let isPro = try await SubscriptionService.shared.restorePurchases()
            restoreResult = RestoreResult(
                success: true,
                message: isPro ? "Pro subscription restored!" : "No active subscription found."
            )
        } catch {
            restoreResult = RestoreResult(
                success: false,
                message: "Failed to restore: \(error.localizedDescription)"
            )
        }
        isRestoring = false
    }
}

struct RestoreResult {
    let success: Bool
    let message: String
}
```
