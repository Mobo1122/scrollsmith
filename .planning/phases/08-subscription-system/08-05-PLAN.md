```swift
import Foundation

/// Usage status from backend /usage endpoint.
struct UsageState: Codable {
    let videosUsed: Int
    let videosLimit: Int  // -1 for unlimited
    let resetDate: Date
    let isPro: Bool
    let canCreateVideo: Bool

    enum CodingKeys: String, CodingKey {
        case videosUsed = "videos_used"
        case videosLimit = "videos_limit"
        case resetDate = "reset_date"
        case isPro = "is_pro"
        case canCreateVideo = "can_create_video"
    }

    /// Human-readable usage string (e.g., "7/10 videos")
    var usageText: String {
        if videosLimit < 0 {
            return "Unlimited"
        }
        return "\(videosUsed)/\(videosLimit) videos"
    }

    /// Progress from 0.0 to 1.0 for progress ring
    var usageProgress: Double {
        if videosLimit <= 0 { return 0 }
        return min(1.0, Double(videosUsed) / Double(videosLimit))
    }

    /// Color tier based on usage (green/yellow/red)
    var usageColor: UsageColor {
        if isPro { return .green }
        switch videosUsed {
        case 0..<7: return .green
        case 7..<10: return .yellow
        default: return .red
        }
    }
}

enum UsageColor {
    case green, yellow, red
}
```

```swift
/// Fetch current usage status.
func getUsage() async throws -> UsageState {
    let url = baseURL.appendingPathComponent("api/v1/usage")

    var request = URLRequest(url: url)
    request.httpMethod = "GET"

    // Add auth header
    if let token = try? await KeychainService.shared.getAccessToken() {
        request.setValue("Bearer \(token)", forHTTPHeaderField: "Authorization")
    }

    let (data, response) = try await URLSession.shared.data(for: request)

    guard let httpResponse = response as? HTTPURLResponse else {
        throw APIError.invalidResponse
    }

    guard httpResponse.statusCode == 200 else {
        throw APIError.httpError(httpResponse.statusCode)
    }

    let decoder = JSONDecoder()
    // Date decoding for reset_date (ISO8601 date only, not datetime)
    let dateFormatter = DateFormatter()
    dateFormatter.dateFormat = "yyyy-MM-dd"
    decoder.dateDecodingStrategy = .formatted(dateFormatter)

    return try decoder.decode(UsageState.self, from: data)
}
```

```swift
import Foundation

/// Manages usage state for the app.
@MainActor
@Observable
final class UsageViewModel {
    static let shared = UsageViewModel()

    private(set) var usage: UsageState?
    private(set) var isLoading = false
    private(set) var error: String?

    private let apiClient = APIClient.shared

    private init() {}

    /// Refresh usage from backend.
    func refreshUsage() async {
        isLoading = true
        error = nil

        do {
            usage = try await apiClient.getUsage()
        } catch {
            self.error = error.localizedDescription
            print("Failed to fetch usage: \(error)")
        }

        isLoading = false
    }

    /// Whether user can create a new video.
    var canCreateVideo: Bool {
        usage?.canCreateVideo ?? true
    }

    /// Whether user is Pro.
    var isPro: Bool {
        usage?.isPro ?? false
    }
}
```

Pattern: Singleton shared instance, @Observable for SwiftUI updates.
