// MARK: - Search DTOs

struct VideoSearchResult: Codable, Identifiable {
    let id: UUID
    let sourceUrl: String?
    let summaryBullets: String?
    let tags: [String]?
    let createdAt: Date
    let rank: Double
    let highlight: String?

    enum CodingKeys: String, CodingKey {
        case id
        case sourceUrl = "source_url"
        case summaryBullets = "summary_bullets"
        case tags
        case createdAt = "created_at"
        case rank, highlight
    }
}

struct VideoSearchResponse: Codable {
    let videos: [VideoSearchResult]
    let query: String
    let total: Int
}

struct VideoListResponse: Codable {
    let videos: [VideoDTO]
    let total: Int
}

// MARK: - Bulk Operation DTOs

struct BulkDeleteResponse: Codable {
    let deletedCount: Int
    let videoIds: [UUID]

    enum CodingKeys: String, CodingKey {

        case videoIds = "video_ids"

// MARK: - APIClient Video Methods

extension APIClient {
    func getVideos(playbookId: UUID? = nil, uncategorized: Bool = false) async throws -> [VideoDTO] {
        var path = "/videos"
        var queryItems: [String] = []

        if let playbookId {
            queryItems.append("playbook_id=\(playbookId.uuidString)")
        }
        if uncategorized {
            queryItems.append("uncategorized=true")
        }

        if !queryItems.isEmpty {
            path += "?" + queryItems.joined(separator: "&")
        }

        let data = try await request(method: "GET", path: path)
        let response = try decoder.decode(VideoListResponse.self, from: data)
        return response.videos
    }

    func searchVideos(query: String, playbookId: UUID? = nil) async throws -> [VideoSearchResult] {
        var path = "/videos/search?q=\(query.addingPercentEncoding(withAllowedCharacters: .urlQueryAllowed) ?? query)"

        if let playbookId {
            path += "&playbook_id=\(playbookId.uuidString)"
        }

        let data = try await request(method: "GET", path: path)
        let response = try decoder.decode(VideoSearchResponse.self, from: data)
        return response.videos
    }

    func bulkDeleteVideos(ids: [UUID]) async throws -> BulkDeleteResponse {
        let body = ["video_ids": ids.map { $0.uuidString }]
        let data = try await request(
            method: "POST",
            path: "/videos/bulk-delete",
            body: try encoder.encode(body)
        )
        return try decoder.decode(BulkDeleteResponse.self, from: data)
    }

    func bulkMoveVideos(ids: [UUID], to playbookId: UUID) async throws -> BulkMoveResponse {
        let body: [String: Any] = [
            "video_ids": ids.map { $0.uuidString },
            "playbook_id": playbookId.uuidString
        ]
        let data = try await request(
            method: "POST",
            path: "/videos/bulk-move",
            body: try JSONSerialization.data(withJSONObject: body)
        )
        return try decoder.decode(BulkMoveResponse.self, from: data)
    }

    func bulkAddToFavorites(ids: [UUID]) async throws -> BulkMoveResponse {
        let body = ["video_ids": ids.map { $0.uuidString }]
        let data = try await request(
            method: "POST",
            path: "/videos/bulk-add-to-favorites",
            body: try encoder.encode(body)
        )
        return try decoder.decode(BulkMoveResponse.self, from: data)
    }

    func deleteVideo(id: UUID) async throws {
        _ = try await request(method: "DELETE", path: "/videos/\(id.uuidString)")
    }

    func updateVideoTags(videoId: UUID, tags: [String]) async throws -> UpdateTagsResponse {
        let body = UpdateTagsRequest(tags: tags)
        let data = try await request(
            method: "PATCH",
            path: "/videos/\(videoId.uuidString)/tags",
            body: try encoder.encode(body)
        )
        return try decoder.decode(UpdateTagsResponse.self, from: data)
    }
}

import Foundation
import SwiftUI

@Observable
final class VideoSelectionManager {
    var isSelecting: Bool = false
    var selectedIds: Set<UUID> = []

    var selectedCount: Int { selectedIds.count }
    var hasSelection: Bool { !selectedIds.isEmpty }

    func toggle(_ id: UUID) {
        if selectedIds.contains(id) {
            selectedIds.remove(id)
        } else {
            selectedIds.insert(id)
        }

        // Exit selection mode if nothing selected
        if selectedIds.isEmpty {
            isSelecting = false
        }
    }

    func enterSelectionMode(with id: UUID) {
        isSelecting = true
        selectedIds.insert(id)
    }

    func clearSelection() {
        selectedIds.removeAll()
        isSelecting = false
    }

    func selectAll(_ ids: [UUID]) {
        isSelecting = true
        selectedIds = Set(ids)
    }

    func isSelected(_ id: UUID) -> Bool {
        selectedIds.contains(id)
    }
}

import Foundation
import SwiftUI

@Observable
final class SearchViewModel {
    var searchText: String = ""
    var debouncedQuery: String = ""
    var isSearching: Bool = false
    var results: [VideoSearchResult] = []
    var error: String?

    private var searchTask: Task<Void, Never>?
    var playbookId: UUID?  // Optional scope to specific Playbook

    // MARK: - Debounced Search

    func onSearchTextChanged(_ newValue: String) {
        searchTask?.cancel()

        if newValue.isEmpty {
            debouncedQuery = ""
            results = []
            return
        }

        searchTask = Task {
            try? await Task.sleep(for: .milliseconds(250))
            guard !Task.isCancelled else { return }

            await MainActor.run {
                debouncedQuery = newValue
            }
        }
    }

    // MARK: - Perform Search

    func performSearch() async {
        guard !debouncedQuery.isEmpty else { return }

        isSearching = true
        error = nil

        do {
            results = try await APIClient.shared.searchVideos(
                query: debouncedQuery,
                playbookId: playbookId
            )
        } catch {
            self.error = error.localizedDescription
            results = []
        }

        isSearching = false
    }

    func clearSearch() {
        searchText = ""
        debouncedQuery = ""
        results = []
        searchTask?.cancel()
    }
}

struct VideoTagEditorView: View {
    let video: VideoDTO
    var onSave: ([String]) -> Void

    @Environment(\.dismiss) private var dismiss
    @State private var tags: [String] = []
    @State private var newTagText: String = ""
    @State private var isSaving: Bool = false
    @State private var error: String?

    var body: some View {
        NavigationStack {
            VStack(alignment: .leading, spacing: 16) {
                // Tag input field
                HStack {

                        .onSubmit {
                            addTag()
                        }

                    Button {
                        addTag()
                    } label: {
                        Image(systemName: "plus.circle.fill")
                            .font(.title2)
                    }
                    .disabled(newTagText.trimmingCharacters(in: .whitespaces).isEmpty)
                }
                .padding(.horizontal)

                // Tag chips
                if tags.isEmpty {
                    ContentUnavailableView(
                        "No Tags",
                        systemImage: "tag.slash",
                        description: Text("Add tags to organize this video")
                    )
                    .frame(maxHeight: 200)
                } else {
                    ScrollView {
                        FlowLayout(spacing: 8) {
                            ForEach(tags, id: \.self) { tag in
                                TagChip(tag: tag) {
                                    removeTag(tag)
                                }
                            }
                        }
                        .padding(.horizontal)
                    }
                }

                Spacer()

                // Error display
                if let error {
                    Text(error)
                        .foregroundStyle(.red)
                        .font(.caption)
                        .padding(.horizontal)
                }
            }
            .padding(.top)
            .navigationTitle("Edit Tags")
            .navigationBarTitleDisplayMode(.inline)
            .toolbar {
                ToolbarItem(placement: .cancellationAction) {
                    Button("Cancel") {
                        dismiss()
                    }
                }
                ToolbarItem(placement: .confirmationAction) {
                    Button("Save") {
                        Task { await saveTags() }
                    }
                    .disabled(isSaving)
                }
            }
            .onAppear {
                tags = video.tags ?? []
            }
        }
        .presentationDetents([.medium, .large])
    }

    // MARK: - Tag Management

    private func addTag() {
        let trimmed = newTagText.trimmingCharacters(in: .whitespaces)
        guard !trimmed.isEmpty else { return }
        guard !tags.contains(trimmed) else {
            newTagText = ""
            return  // Ignore duplicate
        }
        guard tags.count < 20 else {
            error = "Maximum 20 tags allowed"
            return
        }

        tags.append(trimmed)
        newTagText = ""
    }

    private func removeTag(_ tag: String) {
        tags.removeAll { $0 == tag }
    }

    private func saveTags() async {
        isSaving = true
        error = nil

        do {
            let response = try await APIClient.shared.updateVideoTags(videoId: video.id, tags: tags)
            onSave(response.tags)
            dismiss()
        } catch {
            self.error = error.localizedDescription
        }

        isSaving = false
    }
}

// MARK: - Flow Layout for Tags

struct FlowLayout: Layout {
    var spacing: CGFloat = 8

    func sizeThatFits(proposal: ProposedViewSize, subviews: Subviews, cache: inout ()) -> CGSize {
        let result = arrangeSubviews(proposal: proposal, subviews: subviews)
        return result.size
    }

    func placeSubviews(in bounds: CGRect, proposal: ProposedViewSize, subviews: Subviews, cache: inout ()) {
        let result = arrangeSubviews(proposal: proposal, subviews: subviews)

        for (index, frame) in result.frames.enumerated() {
            subviews[index].place(
                at: CGPoint(x: bounds.minX + frame.minX, y: bounds.minY + frame.minY),
                proposal: ProposedViewSize(frame.size)
            )
        }
    }

    private func arrangeSubviews(proposal: ProposedViewSize, subviews: Subviews) -> (size: CGSize, frames: [CGRect]) {
        let maxWidth = proposal.width ?? .infinity
        var currentX: CGFloat = 0
        var currentY: CGFloat = 0
        var lineHeight: CGFloat = 0
        var frames: [CGRect] = []

        for subview in subviews {
            let size = subview.sizeThatFits(.unspecified)

            if currentX + size.width > maxWidth && currentX > 0 {
                currentX = 0
                currentY += lineHeight + spacing
                lineHeight = 0
            }

            frames.append(CGRect(x: currentX, y: currentY, width: size.width, height: size.height))
            lineHeight = max(lineHeight, size.height)
            currentX += size.width + spacing
        }

        let totalHeight = currentY + lineHeight
        return (CGSize(width: maxWidth, height: totalHeight), frames)
    }
}
