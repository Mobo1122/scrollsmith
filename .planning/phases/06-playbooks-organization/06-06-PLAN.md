```swift
import Foundation

// MARK: - Playbook DTOs

struct PlaybookDTO: Codable, Identifiable {
    let id: UUID
    let name: String
    let icon: String?
    let isSystem: Bool
    let videoCount: Int
    let createdAt: Date
    let updatedAt: Date

    enum CodingKeys: String, CodingKey {
        case id, name, icon
        case isSystem = "is_system"
        case videoCount = "video_count"
        case createdAt = "created_at"
        case updatedAt = "updated_at"
    }
}

struct PlaybookListResponse: Codable {
    let playbooks: [PlaybookDTO]
    let total: Int
}

struct CreatePlaybookRequest: Codable {
    let name: String
    let icon: String?
}

struct UpdatePlaybookRequest: Codable {
    let name: String?
    let icon: String?
}

// MARK: - APIClient Playbook Methods

extension APIClient {
    func getPlaybooks() async throws -> [PlaybookDTO] {
        let data = try await request(
            method: "GET",
            path: "/playbooks"
        )
        let response = try decoder.decode(PlaybookListResponse.self, from: data)
        return response.playbooks
    }

    func createPlaybook(name: String, icon: String? = nil) async throws -> PlaybookDTO {
        let body = CreatePlaybookRequest(name: name, icon: icon)
        let data = try await request(
            method: "POST",
            path: "/playbooks",
            body: try encoder.encode(body)
        )
        return try decoder.decode(PlaybookDTO.self, from: data)
    }

    func updatePlaybook(id: UUID, name: String?, icon: String? = nil) async throws -> PlaybookDTO {
        let body = UpdatePlaybookRequest(name: name, icon: icon)
        let data = try await request(
            method: "PATCH",
            path: "/playbooks/\(id.uuidString)",
            body: try encoder.encode(body)
        )
        return try decoder.decode(PlaybookDTO.self, from: data)
    }

    func deletePlaybook(id: UUID) async throws {
        _ = try await request(
            method: "DELETE",
            path: "/playbooks/\(id.uuidString)"
        )
    }

    func assignVideoToPlaybook(videoId: UUID, playbookId: UUID) async throws {
        let body = ["playbook_id": playbookId.uuidString]
        _ = try await request(
            method: "POST",
            path: "/videos/\(videoId.uuidString)/playbooks",
            body: try encoder.encode(body)
        )
    }

    func removeVideoFromPlaybook(videoId: UUID, playbookId: UUID) async throws {
        _ = try await request(
            method: "DELETE",
            path: "/videos/\(videoId.uuidString)/playbooks/\(playbookId.uuidString)"
        )
    }
}
```

```swift
import Foundation
import SwiftUI

@Observable
final class PlaybookViewModel {
    var playbooks: [PlaybookDTO] = []
    var isLoading: Bool = false
    var error: String?
    var lastSelectedPlaybookId: UUID?  // "Remember last selected" behavior

    // MARK: - Load Playbooks

    func loadPlaybooks() async {
        isLoading = true
        error = nil

        do {
            playbooks = try await APIClient.shared.getPlaybooks()
        } catch {
            self.error = error.localizedDescription
        }

        isLoading = false
    }

    // MARK: - Create Playbook

    func createPlaybook(name: String, icon: String? = nil) async -> Bool {
        do {
            let newPlaybook = try await APIClient.shared.createPlaybook(name: name, icon: icon)
            playbooks.insert(newPlaybook, at: playbooks.first?.isSystem == true ? 1 : 0)
            return true
        } catch {
            self.error = error.localizedDescription
            return false
        }
    }

    // MARK: - Update Playbook

    func updatePlaybook(id: UUID, name: String) async -> Bool {
        do {
            let updated = try await APIClient.shared.updatePlaybook(id: id, name: name)
            if let index = playbooks.firstIndex(where: { $0.id == id }) {
                playbooks[index] = updated
            }
            return true
        } catch {
            self.error = error.localizedDescription
            return false
        }
    }

    // MARK: - Delete Playbook

    func deletePlaybook(id: UUID) async -> Bool {
        do {
            try await APIClient.shared.deletePlaybook(id: id)
            playbooks.removeAll { $0.id == id }
            return true
        } catch {
            self.error = error.localizedDescription
            return false
        }
    }

    // MARK: - Assign Video

    func assignVideo(_ videoId: UUID, to playbookId: UUID) async -> Bool {
        do {
            try await APIClient.shared.assignVideoToPlaybook(videoId: videoId, playbookId: playbookId)
            lastSelectedPlaybookId = playbookId  // Remember selection
            await loadPlaybooks()  // Refresh counts
            return true
        } catch {
            self.error = error.localizedDescription
            return false
        }
    }

    // MARK: - Helper

    var favoritesPlaybook: PlaybookDTO? {
        playbooks.first { $0.isSystem && $0.name == "Favorites" }
    }

    var userPlaybooks: [PlaybookDTO] {
        playbooks.filter { !$0.isSystem }
    }
}
```

```swift
import SwiftUI

struct PlaybookListView: View {
    @State private var viewModel = PlaybookViewModel()
    @State private var showCreateSheet = false
    @State private var newPlaybookName = ""
    @State private var editingPlaybook: PlaybookDTO?

    var body: some View {
        NavigationStack {
            Group {
                if viewModel.isLoading && viewModel.playbooks.isEmpty {
                    ProgressView("Loading...")
                } else if let error = viewModel.error, viewModel.playbooks.isEmpty {
                    ContentUnavailableView("Error", systemImage: "exclamationmark.triangle", description: Text(error))
                } else {
                    List {
                        // Favorites at top
                        if let favorites = viewModel.favoritesPlaybook {
                            NavigationLink(destination: PlaybookDetailView(playbook: favorites)) {
                                PlaybookRow(playbook: favorites)
                            }
                        }

                        // User Playbooks
                        Section("My Playbooks") {
                            ForEach(viewModel.userPlaybooks) { playbook in
                                NavigationLink(destination: PlaybookDetailView(playbook: playbook)) {
                                    PlaybookRow(playbook: playbook)
                                }
                                .swipeActions(edge: .trailing) {
                                    Button("Delete", role: .destructive) {
                                        Task { await viewModel.deletePlaybook(id: playbook.id) }
                                    }
                                    Button("Edit") {
                                        editingPlaybook = playbook
                                    }
                                    .tint(.blue)
                                }
                            }
                        }
                    }
                }
            }
            .navigationTitle("Playbooks")
            .toolbar {
                ToolbarItem(placement: .topBarTrailing) {
                    Button {
                        showCreateSheet = true
                    } label: {
                        Image(systemName: "plus")
                    }
                }
            }
            .refreshable {
                await viewModel.loadPlaybooks()
            }
            .task {
                await viewModel.loadPlaybooks()
            }
            .sheet(isPresented: $showCreateSheet) {
                CreatePlaybookSheet(viewModel: viewModel, isPresented: $showCreateSheet)
            }
            .sheet(item: $editingPlaybook) { playbook in
                EditPlaybookSheet(viewModel: viewModel, playbook: playbook, isPresented: Binding(
                    get: { editingPlaybook != nil },
                    set: { if !$0 { editingPlaybook = nil } }
                ))
            }
        }
    }
}

                Text(playbook.name)
                    .font(.headline)
                Text("\(playbook.videoCount) videos")
                    .font(.caption)
                    .foregroundStyle(.secondary)
            }
        }
    }
}

}
```

```swift
import SwiftUI

struct PlaybookDetailView: View {
    let playbook: PlaybookDTO
    @State private var videos: [VideoDTO] = []
    @State private var isLoading = false
    @State private var error: String?

    var body: some View {
        Group {
            if isLoading && videos.isEmpty {
                ProgressView("Loading videos...")
            } else if videos.isEmpty {
                ContentUnavailableView(
                    "No Videos",
                    systemImage: "video.slash",
                    description: Text("Add videos to this Playbook from the video picker")
                )
            } else {
                ScrollView {
                    LazyVGrid(columns: [GridItem(.adaptive(minimum: 150))], spacing: 16) {
                        ForEach(videos) { video in
                            VideoThumbnailCard(video: video)
                        }
                    }
                    .padding()
                }
            }
        }
        .navigationTitle(playbook.name)
        .task {
            await loadVideos()
        }
        .refreshable {
            await loadVideos()
        }
    }

    private func loadVideos() async {
        isLoading = true
        do {
            // Uses GET /videos?playbook_id=xxx from Plan 04
            videos = try await APIClient.shared.getVideos(playbookId: playbook.id)
        } catch {
            self.error = error.localizedDescription
        }
        isLoading = false
    }
}
```

```swift
import SwiftUI

struct PlaybookPickerSheet: View {
    @Bindable var viewModel: PlaybookViewModel
    let videoId: UUID
    var onDismiss: () -> Void

    @Environment(\.dismiss) private var dismiss
    @State private var showAllPlaybooks = false

    var body: some View {
        VStack(spacing: 16) {
            // Header
            HStack {
                Text("Add to Playbook")
                    .font(.headline)
                Spacer()
                Button {
                    onDismiss()
                    dismiss()
                } label: {
                    Image(systemName: "xmark.circle.fill")
                        .foregroundStyle(.secondary)
                }
            }
            .padding(.horizontal)
            .padding(.top, 8)

            // Horizontal scrolling chips
            ScrollView(.horizontal, showsIndicators: false) {
                HStack(spacing: 12) {
                    // Favorites first
                    if let favorites = viewModel.favoritesPlaybook {
                        PlaybookChip(
                            playbook: favorites,
                            isSelected: viewModel.lastSelectedPlaybookId == favorites.id
                        ) {
                            Task {
                                if await viewModel.assignVideo(videoId, to: favorites.id) {
                                    onDismiss()
                                    dismiss()
                                }
                            }
                        }
                    }

                    // User Playbooks
                    ForEach(viewModel.userPlaybooks) { playbook in
                        PlaybookChip(
                            playbook: playbook,
                            isSelected: viewModel.lastSelectedPlaybookId == playbook.id
                        ) {
                            Task {
                                if await viewModel.assignVideo(videoId, to: playbook.id) {
                                    onDismiss()
                                    dismiss()
                                }
                            }
                        }
                    }
                }
                .padding(.horizontal)
            }

            // See all button
            Button {
                showAllPlaybooks = true
            } label: {
                Text("See all playbooks")
                    .font(.subheadline)
            }
            .padding(.bottom, 8)
        }
        .presentationDetents([.height(160), .medium])
        .presentationDragIndicator(.visible)
        .task {
            if viewModel.playbooks.isEmpty {
                await viewModel.loadPlaybooks()
            }
        }
        .sheet(isPresented: $showAllPlaybooks) {
            NavigationStack {
                List(viewModel.playbooks) { playbook in
                    Button {
                        Task {
                            if await viewModel.assignVideo(videoId, to: playbook.id) {
                                showAllPlaybooks = false
                                onDismiss()
                                dismiss()
                            }
                        }
                    } label: {
                        PlaybookRow(playbook: playbook)
                    }
                }
                .navigationTitle("All Playbooks")
                .navigationBarTitleDisplayMode(.inline)
                .toolbar {
                    ToolbarItem(placement: .cancellationAction) {
                        Button("Cancel") { showAllPlaybooks = false }
                    }
                }
            }
        }
    }
}

struct PlaybookChip: View {
    let playbook: PlaybookDTO
    let isSelected: Bool
    let action: () -> Void

    var body: some View {
        Button(action: action) {
            HStack {
                Image(systemName: playbook.icon ?? "folder")
                Text(playbook.name)
            }
            .font(.subheadline)
            .padding(.horizontal, 16)
            .padding(.vertical, 10)
            .background(isSelected ? Color.blue : Color.gray.opacity(0.2))
            .foregroundStyle(isSelected ? .white : .primary)
            .clipShape(Capsule())
        }
    }
}
```
