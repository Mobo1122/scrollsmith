3. Create `ios/Scrollsmith/Services/AnalyticsService.swift`:
   ```swift
   import Foundation
   import Mixpanel

   /// Centralized analytics service using Mixpanel.
   ///
   /// Tracks key metrics per phase success criteria:
   /// - Video views
   /// - Habit completions
   /// - Free -> Pro conversions
   /// - User retention (via identify)
   ///
   /// Mixpanel doesn't use IDFA, so no ATT prompt required.
   actor AnalyticsService {
       static let shared = AnalyticsService()

       private var isConfigured = false

       private init() {}

       /// Configure Mixpanel SDK.
       ///
       /// Call this once at app launch.
       /// Safe to call multiple times (subsequent calls are no-ops).
       func configure() {
           guard !isConfigured else { return }

           // Skip if no token configured
           guard !Configuration.mixpanelToken.isEmpty else {
               print("Analytics: Mixpanel token not configured, skipping")
               return
           }

           Mixpanel.initialize(
               token: Configuration.mixpanelToken,
               trackAutomaticEvents: false  // We track specific events only
           )

           isConfigured = true
           print("Analytics: Mixpanel initialized")
       }

       // MARK: - User Identification

       /// Identify user for retention tracking.
       ///
       /// Call when user logs in. Associates all events with this user.
       /// - Parameters:
       ///   - userId: The user's UUID
       ///   - isPro: Whether user has Pro subscription
       func identifyUser(userId: UUID, isPro: Bool) {
           guard isConfigured else { return }

           Mixpanel.mainInstance().identify(distinctId: userId.uuidString)
           Mixpanel.mainInstance().people.set(properties: [
               "is_pro": isPro,
               "platform": "ios",
               "app_version": Configuration.appVersion,
           ])

           print("Analytics: Identified user \(userId.uuidString.prefix(8))...")
       }

       /// Reset user identity on logout.
       func resetUser() {
           guard isConfigured else { return }
           Mixpanel.mainInstance().reset()
           print("Analytics: User reset")
       }

       /// Update user's Pro status.
       ///
       /// Call when subscription status changes.
       func updateProStatus(isPro: Bool) {
           guard isConfigured else { return }
           Mixpanel.mainInstance().people.set(properties: [
               "is_pro": isPro,
           ])
       }

       // MARK: - Key Events

       /// Track video view.
       ///
       /// Call when user opens a video's summary/detail view.
       /// - Parameter videoId: The viewed video's UUID
       func trackVideoView(videoId: UUID) {
           guard isConfigured else { return }

           Mixpanel.mainInstance().track(event: "video_viewed", properties: [
               "video_id": videoId.uuidString,
           ])
       }

       /// Track habit completion.
       ///
       /// Call when user marks a habit as complete.
       /// - Parameters:
       ///   - habitId: The completed habit's UUID
       ///   - streakLength: Current streak after completion
       func trackHabitCompletion(habitId: UUID, streakLength: Int) {
           guard isConfigured else { return }

           Mixpanel.mainInstance().track(event: "habit_completed", properties: [
               "habit_id": habitId.uuidString,
               "streak_length": streakLength,
           ])
       }

       /// Track Pro subscription conversion.
       ///
       /// Call when user successfully purchases Pro.
       /// - Parameter source: Where the conversion happened (e.g., "paywall", "habit_limit", "format_upgrade")
       func trackProConversion(source: String) {
           guard isConfigured else { return }

           Mixpanel.mainInstance().track(event: "pro_conversion", properties: [
               "source": source,
           ])

           // Also update user property
           Mixpanel.mainInstance().people.set(properties: [
               "is_pro": true,
               "converted_at": ISO8601DateFormatter().string(from: Date()),
           ])

           print("Analytics: Pro conversion from \(source)")
       }

       /// Track paywall shown (for conversion funnel).
       ///
       /// - Parameter source: What triggered the paywall
       func trackPaywallShown(source: String) {
           guard isConfigured else { return }

           Mixpanel.mainInstance().track(event: "paywall_shown", properties: [
               "source": source,
           ])
       }

       /// Track paywall dismissed without purchase.
       ///
       /// - Parameter source: What triggered the paywall
       func trackPaywallDismissed(source: String) {
           guard isConfigured else { return }

           Mixpanel.mainInstance().track(event: "paywall_dismissed", properties: [
               "source": source,
           ])
       }

       // MARK: - Screen Tracking

       /// Track screen view for retention analysis.
       ///
       /// - Parameter screenName: Name of the screen
       func trackScreen(_ screenName: String) {
           guard isConfigured else { return }

           Mixpanel.mainInstance().track(event: "screen_viewed", properties: [
               "screen_name": screenName,
           ])
       }
   }
   ```
